## **The Critical Database Schema**

```sql
-- Main table stores everything, but clearly distinguished
CREATE TABLE knowledge_units (
    id VARCHAR(36) PRIMARY KEY,
    content JSON NOT NULL,
    
    -- PROVENANCE TIER (most important)
    source_tier VARCHAR(20) NOT NULL CHECK (
        source_tier IN ('primary_source', 'synthesis', 'pedagogical_aid', 'system_inference')
    ),
    
    -- HOW IT CAME TO BE
    provenance_type VARCHAR(30) NOT NULL CHECK (
        provenance_type IN (
            'extracted_manual', 'extracted_llm',
            'authored_manual', 'authored_llm', 
            'system_generated'
        )
    ),
    
    -- AUTHORSHIP (NULL for extractions, set for authored content)
    author_id VARCHAR(36) NULL,
    author_role VARCHAR(20) NULL,  -- 'teacher', 'researcher', 'curator'
    
    -- SOURCE LINKS (for everything)
    based_on_sources JSON,  -- Array of source IDs this is based on
    
    -- QUALITY METRICS
    factual_accuracy FLOAT DEFAULT 1.0,
    pedagogical_value FLOAT DEFAULT 1.0,
    confidence FLOAT DEFAULT 1.0,
    
    -- USAGE TRACKING
    times_used_in_training INT DEFAULT 0,  -- For LLM training
    times_served_to_students INT DEFAULT 0,
    
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    
    INDEX idx_source_tier (source_tier),
    INDEX idx_provenance (provenance_type),
    INDEX idx_author (author_id)
);
```

## **The Training Data Firewall**

```go
// Function that serves data for LLM training
func getTrainingData() []TrainingExample {
    // ONLY primary source extractions
    query := `
        SELECT * FROM knowledge_units 
        WHERE source_tier = 'primary_source' 
          AND provenance_type IN ('extracted_manual', 'extracted_llm')
          AND confidence > 0.85
          AND times_used_in_training < 10  // Prevent overfitting
        ORDER BY RAND() 
        LIMIT 1000
    `
    
    // Explicitly EXCLUDE:
    // - source_tier = 'synthesis' (your narratives)
    // - source_tier = 'pedagogical_aid' (your teaching content)
    // - source_tier = 'system_inference' (generated content)
    // - provenance_type = 'authored_*' (anything authored)
    
    return executeQuery(query)
}
```

## **The Student Delivery with Provenance**

```go
func getStudentContent(topic string, studentLevel string) []ContentItem {
    // Get ALL content types, but with provenance labels
    query := `
        SELECT 
            content,
            CASE source_tier
                WHEN 'primary_source' THEN 'Source Fact'
                WHEN 'synthesis' THEN 'Learning Story'
                WHEN 'pedagogical_aid' THEN 'Teaching Aid'
                WHEN 'system_inference' THEN 'System Insight'
            END as provenance_label,
            
            CASE 
                WHEN author_id IS NOT NULL THEN CONCAT('By ', author_id)
                WHEN provenance_type = 'extracted_manual' THEN 'From textbook'
                ELSE 'System generated'
            END as attribution,
            
            factual_accuracy,
            pedagogical_value
        FROM knowledge_units
        WHERE topic = ? 
          AND appropriate_for_level(?, studentLevel)
        ORDER BY 
            source_tier ASC,  -- Primary sources first
            pedagogical_value DESC
    `
    
    return executeQuery(query, topic, studentLevel)
}
```
